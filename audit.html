Verify it yourself with Python 
import json, hashlib
d = json.load(open("ravenshield-evidence.json", encoding="utf-8"))
prev = "0" * 64
for i, r in enumerate(d["records"]):
    body = [r["seq"], r["id"], r["ts"], r["source"], r["kind"],
            r["severity"], r["title"], r.get("detail"), r["prev"]]
    h = hashlib.sha256(json.dumps(body, separators=(",", ":"),
                       ensure_ascii=False).encode()).hexdigest()
    assert r["prev"] == prev and r["seq"] == i and h == r["hash"], f"broken at {i}"
    prev = h
print("chain OK, head", prev)
